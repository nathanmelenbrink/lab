Week 6 Lab: Useless Box Code



#include <ESP32Servo.h>

int switchPin = 35;
int switchServoPin = 32;

class Switch{
  int currentState;
  int lastState = 0;

  public:
  Switch(int switchPin){
    pinMode(switchPin, INPUT_PULLUP);
  };

  bool isPressed(){
    currentState = digitalRead(switchPin);
    if (currentState == HIGH && lastState == LOW){
      lastState = currentState;
      return true;
    }
    else if (currentState == LOW && lastState == HIGH){
      lastState = currentState;
      return false;
    }
  }
};

class UselessServo{
  Servo servo;
  int pos;
  int increment;
  int updateInterval;
  unsigned long lastUpdate;
  int startPos;
  int endPos;

  public:
  UselessServo(int interval, int start, int target){
    updateInterval = interval;
    increment = -1;
    startPos = start-1;
    pos = startPos;
    endPos = target;
  }

  void Attach(int pin){
    servo.attach(pin);
    servo.write(startPos);
  }

  void Detach(){
    servo.detach();
  }

  void toTarget(){
    if((millis() - lastUpdate) > updateInterval && pos != endPos){
      lastUpdate = millis();
      pos += increment;
      servo.write(pos);
    }
  }

  void Return(){
    if((millis() - lastUpdate) > updateInterval && pos != startPos){
      lastUpdate = millis();
      pos -= increment;
      servo.write(pos);
    }
  }
};

Switch mainSwitch(switchPin);
UselessServo switchServo(0.5, 180, 0);

void setup() {
  switchServo.Attach(switchServoPin);
  Serial.begin(9600);
}

void loop() {
  if(mainSwitch.isPressed() == 1){
    switchServo.toTarget();

  }
  else{
    switchServo.Return();
  }
}
